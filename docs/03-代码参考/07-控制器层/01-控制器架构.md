## 📋 概述

Controllers控制器层负责接收HTTP请求、调用Service处理业务逻辑、返回统一格式的响应。

### 设计原则

```
✅ 薄控制器 - 不包含业务逻辑
✅ 统一响应 - 使用ApiResponse统一格式
✅ 请求验证 - 使用FormRequest验证
✅ 资源转换 - 使用Resource格式化输出
✅ RESTful - 遵循REST API设计规范
```

## 💡 Controller开发最佳实践

### 1. 标准Controller结构

```php
<?php

namespace App\Modules\{Module}\Controllers;

use App\Infrastructure\Http\Controllers\BaseController;
use App\Modules\{Module}\Services\{Module}Service;
use App\Modules\{Module}\Requests\{Action}Request;
use App\Modules\{Module}\Resources\{Module}Resource;
use Illuminate\Http\JsonResponse;

/**
 * {Module} Controller
 * 
 * {模块}API控制器
 * 
 * @version 1.0.0
 * @date 2025-11-01
 */
class {Module}Controller extends BaseController
{
    protected {Module}Service $service;
    
    public function __construct({Module}Service $service)
    {
        $this->service = $service;
    }

    /**
     * 获取列表
     */
    public function index(Request $request): JsonResponse
    {
        $data = $this->service->getList($request->all());
        return $this->success($data, '获取成功');
    }

    /**
     * 获取详情
     */
    public function show(int $id): JsonResponse
    {
        $data = $this->service->getDetail($id);
        
        if (!$data) {
            return $this->error('资源不存在', 404);
        }
        
        return $this->success($data, '获取成功');
    }

    /**
     * 创建资源
     */
    public function store({Action}Request $request): JsonResponse
    {
        $data = $this->service->create($request->validated());
        return $this->success($data, '创建成功', 201);
    }

    /**
     * 更新资源
     */
    public function update(int $id, {Action}Request $request): JsonResponse
    {
        $data = $this->service->update($id, $request->validated());
        return $this->success($data, '更新成功');
    }

    /**
     * 删除资源
     */
    public function destroy(int $id): JsonResponse
    {
        $this->service->delete($id);
        return $this->success(null, '删除成功');
    }
}
```

### 3. FormRequest验证

```php
✅ 正确做法 - 使用FormRequest:
public function store(StoreExerciseRequest $request): JsonResponse
{
    // $request->validated() 已经通过验证
    $data = $this->service->create($request->validated());
    return $this->success($data);
}

❌ 错误做法 - 手动验证:
public function store(Request $request): JsonResponse
{
    $validated = $request->validate([  // ❌ 不要在Controller中验证
        'name' => 'required|string',
        ...
    ]);
}
```

---

### 4. Resource格式化

```php
✅ 使用Resource转换:
public function show(int $id): JsonResponse
{
    $exercise = $this->service->getDetail($id);
    return $this->success(new ExerciseResource($exercise));
}

// Collection转换
public function index(): JsonResponse
{
    $exercises = $this->service->getList();
    return $this->success(ExerciseResource::collection($exercises));
}
```

### 5. 异常处理

```php
✅ 让全局异常处理器处理:
public function show(int $id): JsonResponse
{
    // 如果抛出异常，会被全局异常处理器捕获
    $data = $this->service->getDetail($id);
    return $this->success($data);
}

// 业务异常示例

    throw new ResourceNotFoundException('动作不存在');
}

// 全局异常处理器会返回统一格式:
// {
//   "code": 404,
//   "msg": "动作不存在",
```

---

### 2. 权限检查

```php
// 使用中间件检查会员权限
Route::middleware(['auth:sanctum', 'membership:energy'])->group(function () {
    Route::post('/training/ai-recommend', [TrainingController::class, 'aiRecommend']);
});

// 或在Controller中手动检查
public function aiRecommend(): JsonResponse
{
    $userId = auth()->id();
    
    if (!$this->membershipService->checkPermission($userId, 'ai_recommendation')) {
        return $this->error('需要升级会员才能使用此功能', 403);
    }
    
    // 执行AI推荐
    $result = $this->trainingService->getAIRecommendation($userId);
    return $this->success($result);
}
```

## 📊 统一响应格式

### ApiResponse使用

```php
// 成功响应
return $this->success($data, '操作成功', 200);

// 错误响应
return $this->error('操作失败', 400);

// 分页响应
return $this->paginate($data, [
    'current_page' => 1,
    'total' => 100,
    'per_page' => 20,
    'last_page' => 5
]);

// 资源创建成功
return $this->success($data, '创建成功', 201);

// 无内容响应
return $this->success(null, '删除成功', 204);
```

### 响应格式标准

```json
// 成功响应
{
  "code": 200,
  "msg": "操作成功",
  "data": {...}
}

// 错误响应
{
  "code": 400,
  "msg": "操作失败",
  "data": null,
  "errors": {...}  // 验证错误时包含详情
}

// 分页响应
{
  "code": 200,
  "msg": "获取成功",
  "data": {
    "rows": [...],
    "pagination": {
      "current_page": 1,
      "total": 100,
      "per_page": 20,
```

## 🎯 RESTful API设计

### 标准RESTful端点

| HTTP方法 | 路由 | Action | 说明 |
|---------|------|--------|------|
| GET | /api/resources | index | 获取资源列表 |
| GET | /api/resources/{id} | show | 获取单个资源 |
| POST | /api/resources | store | 创建资源 |
| PUT | /api/resources/{id} | update | 更新资源（完整） |
| PATCH | /api/resources/{id} | update | 更新资源（部分） |
| DELETE | /api/resources/{id} | destroy | 删除资源 |

### 自定义操作端点

```php
// ✅ 推荐：POST + 动词
POST /api/training/plans/{id}/complete  // 完成训练计划
POST /api/membership/orders/{id}/cancel // 取消订单
POST /api/users/password/reset          // 重置密码

// ❌ 不推荐：GET + 动词
GET /api/training/plans/{id}/complete   // 不安全
GET /api/users/logout                   // 应该用POST
```

```php
use Tests\TestCase;

class ExerciseControllerTest extends TestCase

                             'difficulty' => 'beginner'
                         ]);
        
        $response->assertStatus(201)
                 ->assertJsonPath('data.name', 'Test Exercise');
    }
```

## 📝 Controller命名规范

### 标准CRUD方法

```php
✅ 正确命名:
public function index()     // 列表
public function show($id)   // 详情
public function store()     // 创建
public function update($id) // 更新
public function destroy($id)// 删除

❌ 错误命名:
public function getList()   // ❌ 应该用 index
public function getDetail() // ❌ 应该用 show
public function add()       // ❌ 应该用 store
public function delete()    // ❌ 应该用 destroy
```
