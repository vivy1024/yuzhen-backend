## 1. 概述

### ⚠️ 重要声明

**当前状态**: 会员系统已实现但**处于禁用状态**（`MEMBERSHIP_SYSTEM_ENABLED=false`）

**禁用原因**:
- 这是个人开发项目，开发者没有企业资质
- 个人ICP备案不允许涉及经营性业务（付费会员属于经营性业务）
- 当前采用**打赏模式**替代会员系统，所有用户享受统一服务
- 数据库中的定价层级（¥3/月、¥8/月）仅为技术实现，**未激活且不可购买**

**未来计划**:
- 获得个人独资企业资质后，可通过环境变量一键启用会员系统
- 启用前需完成企业ICP备案变更

### 1.1 系统架构

会员系统采用分层架构设计，包含以下核心组件：

```
Controller层（API接口）
    ↓
Service层（业务逻辑）
    ↓
Repository层（数据访问）
    ↓
Model层（数据模型）
```

### 1.2 核心功能

- ✅ **会员等级管理** - 数据库包含免费版、暖心会员、能量会员（**当前未激活**）
- ✅ **订单管理** - 订单创建、支付截图上传、审核流程（**当前禁用**）
- ✅ **权限控制** - 基于会员等级的功能访问控制
- ✅ **配置开关** - 支持启用/禁用会员系统（**当前禁用以符合合规要求**）
- ✅ **用量追踪** - DAG模式和Agent模式分开统计
- ✅ **打赏奖励** - 管理员可手动添加额外次数（**当前使用中**）

### 1.3 技术栈

- **框架**: Laravel 11
- **数据库**: MySQL 8.0
- **缓存**: Redis（可选，支持file驱动）
- **认证**: Laravel Sanctum
- **事件系统**: Laravel Events
- **部署**: Zeabur云平台（阿里云北京区域）

```php
    }

public function getFeatures(): array

    if ($this->unlock_all_exercises) {
        $features[] = '解锁所有动作库';

    if ($this->max_training_plans > 10) {
        $features[] = '无限训练计划';
    } else {
        $features[] = "最多{$this->max_training_plans}个训练计划";
    }
    
    if ($this->ai_recommendation) {
        $features[] = 'AI智能推荐';
    }
    
    if ($this->data_analysis) {
        $features[] = '数据分析报表';
    }
    
    if ($this->coach_service) {
        $features[] = '专属教练服务';
    }
    
    return $features;
}
```

```php
    $this->save();

public function cancel(): void
{
    $this->is_active = 0;
    $this->auto_renew = false;
    $this->save();
}

public function scopeActive($query)
{
    return $query->where('is_active', 1)
                 ->where('expires_at', '>', now());

public function scopeExpiringSoon($query)
{
    return $query->where('is_active', 1)
                 ->whereBetween('expires_at', [now(), now()->addDays(7)]);
```

```php
public function checkPermission(int $userId, string $feature): bool
{
    $userMembership = $this->getUserMembership($userId);
    
    if (!$userMembership) {
        // 没有会员，只有免费权限
        return $this->checkFreePermission($feature);
    }

    return match($feature) {
        'unlock_all_exercises' => $membership['unlock_all_exercises'] ?? false,
        'ai_recommendation' => $membership['ai_recommendation'] ?? false,
        'data_analysis' => $membership['data_analysis'] ?? false,
```

```php
public function handlePaymentSuccess(int $orderId, string $tradeNo, string $payMethod): bool
{
    return DB::transaction(function () use ($orderId, $tradeNo, $payMethod) {
        // 1. 更新订单状态
        $order = $this->orderRepo->findById($orderId);
        if (!$order || $order['status'] !== 'pending') {
            throw new \Exception('订单状态异常');
        }

        $this->orderRepo->markAsPaid($orderId, $tradeNo, $payMethod);

        // 2. 创建或更新用户会员
        $membership = $this->membershipRepo->findById($order['membership_id']);
        $this->activateUserMembership(
            $order['user_id'], 
            $order['membership_id'], 
            $membership['duration_days'], 
            $orderId
        );

        // 3. 触发事件
        Event::dispatch(new MembershipPurchased($order['user_id'], $order['membership_id']));

        return true;
    });
}
```

```php
public function checkLimit(int $userId, string $strategy = 'dag'): array
{
    $today = Carbon::today()->toDateString();
    
    // 获取或创建今日统计
    $stats = $this->getOrCreateDailyStats($userId, $today);
    $bonusCredits = $this->getBonusCredits($userId);
    
    if ($strategy === 'agent') {
        $used = $stats->agent_queries;
        $limit = $stats->agent_limit;
        $bonusTotal = $bonusCredits->total_agent_credits ?? 0;
        $bonusUsed = $bonusCredits->used_agent_credits ?? 0;
    } else {
        $used = $stats->dag_queries;
        $limit = $stats->dag_limit;
        $bonusTotal = $bonusCredits->total_dag_credits ?? 0;
        $bonusUsed = $bonusCredits->used_dag_credits ?? 0;
    }
    
    // 计算剩余次数
    $bonusRemaining = max(0, $bonusTotal - $bonusUsed);
    $dailyRemaining = $limit == -1 ? PHP_INT_MAX : max(0, $limit - $used);
    $totalRemaining = $dailyRemaining + $bonusRemaining;
    
    // 无限制
    if ($limit == -1) {
        return [
            'allowed' => true,
            'remaining' => -1,
            'limit' => -1,
            'used' => $used,
            'bonus_remaining' => $bonusRemaining,
            'message' => '无限制'
        ];
    }
    
    // 检查是否允许
    $allowed = $totalRemaining > 0;
    
    $message = $allowed 
        ? "今日剩余 {$dailyRemaining} 次" . ($bonusRemaining > 0 ? "，额外 {$bonusRemaining} 次" : "")
        : "今日次数已用完，打赏可获得更多次数";
    
    return [
        'allowed' => $allowed,
        'remaining' => $totalRemaining,
        'daily_remaining' => $dailyRemaining,
        'bonus_remaining' => $bonusRemaining,
        'limit' => $limit,
        'used' => $used,
        'message' => $message
    ];
}
```

```php
public function recordQuery(int $userId, string $strategy = 'dag', array $metadata = []): bool
{
    $today = Carbon::today()->toDateString();
    
    try {
        DB::beginTransaction();
        
        // 获取或创建今日统计
        $stats = $this->getOrCreateDailyStats($userId, $today);
        $bonusCredits = $this->getBonusCredits($userId);

        if ($strategy === 'agent') {
            $dailyRemaining = $stats->agent_limit == -1 ? PHP_INT_MAX : max(0, $stats->agent_limit - $stats->agent_queries);
            $bonusRemaining = max(0, ($bonusCredits->total_agent_credits ?? 0) - ($bonusCredits->used_agent_credits ?? 0));
            
            if ($dailyRemaining > 0) {
                // 使用每日次数
                DB::table('user_usage_stats')
                    ->where('id', $stats->id)
                    ->increment('agent_queries');
            } elseif ($bonusRemaining > 0) {
                // 使用额外次数
                DB::table('user_bonus_credits')
                    ->where('user_id', $userId)
                    ->increment('used_agent_credits');
            } else {
                DB::rollBack();
                return false;
            }
        } else {
            // DAG模式逻辑类似
            // ...
        }
        
        DB::commit();
        $this->clearCache($userId);
        
        return true;
        
    } catch (\Exception $e) {
        DB::rollBack();
        Log::error("用量记录失败", ['user_id' => $userId, 'error' => $e->getMessage()]);
        return false;
    }
}
```

### 7.1 系统概述

用量追踪系统记录用户使用AI对话的次数，分为两种模式：
- **DAG模式**: 使用预定义的DAG模板进行对话
- **Agent模式**: 使用AI Agent进行复杂对话

### 7.2 数据流程

## 8. 打赏奖励机制

### 8.1 奖励规则

管理员可以根据用户打赏金额手动添加额外次数：

| 打赏金额 | DAG额外次数 | Agent额外次数 |
|---------|------------|--------------|
| ¥5 | 5次 | 1次 |
| ¥10 | 10次 | 3次 |
| ¥20 | 25次 | 8次 |
| ¥50 | 70次 | 20次 |

### 8.2 管理员操作

```php
$service = app(UsageTrackingService::class);

// 用户打赏¥10后，添加额外次数
$success = $service->addBonusCredits(
    userId: $userId,
    dagCredits: 10,
    agentCredits: 3,
    reason: '打赏¥10'
);

if ($success) {
    // 通知用户
    Notification::send($userId, new BonusCreditsAddedNotification(10, 3));
}
```

### 8.3 额外次数使用逻辑

系统会优先使用每日限制次数，用完后自动使用额外次数：

```
1. 检查每日限制是否还有剩余
   ↓ 有剩余
   使用每日次数
   
   ↓ 无剩余
2. 检查额外次数是否还有剩余
   ↓ 有剩余
   使用额外次数
   
   ↓ 无剩余
3. 返回限制提示
```

```php
// 在AI对话接口中使用
public function chat(Request $request)
{
    $userId = auth()->id();
    $strategy = $request->input('strategy', 'dag'); // dag 或 agent
    
    $usageService = app(UsageTrackingService::class);
    
    // 1. 检查限制
    $limitCheck = $usageService->checkLimit($userId, $strategy);
    
    if (!$limitCheck['allowed']) {
        return response()->json([
            'code' => 429,
            'msg' => $limitCheck['message'],
            'data' => [
                'remaining' => $limitCheck['remaining'],
                'daily_remaining' => $limitCheck['daily_remaining'],
                'bonus_remaining' => $limitCheck['bonus_remaining']
            ]
        ], 429);
    }
    
    // 2. 执行对话
    $response = $this->executeChat($request);
    
    // 3. 记录用量
    $usageService->recordQuery($userId, $strategy, [
        'query' => $request->input('query'),
        'dag_template' => $request->input('dag_template'),
        'response_time' => $response['time']
    ]);
    
    return response()->json($response);
}
```

### 10.4 打赏奖励示例

```php
// 管理员处理打赏
public function handleDonation(Request $request)
{
    $userId = $request->input('user_id');
    $amount = $request->input('amount');
    
    // 根据打赏金额计算额外次数
    $credits = match(true) {
        $amount >= 50 => ['dag' => 70, 'agent' => 20],
        $amount >= 20 => ['dag' => 25, 'agent' => 8],
        $amount >= 10 => ['dag' => 10, 'agent' => 3],
        $amount >= 5 => ['dag' => 5, 'agent' => 1],
        default => ['dag' => 0, 'agent' => 0]
    };
    
    $usageService = app(UsageTrackingService::class);
    
    $success = $usageService->addBonusCredits(
        userId: $userId,
        dagCredits: $credits['dag'],
        agentCredits: $credits['agent'],
        reason: "打赏¥{$amount}"
    );
    
    if ($success) {
        // 通知用户
        Notification::send($userId, new BonusCreditsAddedNotification(
            $credits['dag'],
            $credits['agent']
        ));
        
        return response()->json([
            'code' => 200,
            'msg' => '额外次数添加成功',
            'data' => $credits
        ]);
    }
    
    return response()->json([
        'code' => 500,
        'msg' => '添加失败'
    ], 500);
}
```
